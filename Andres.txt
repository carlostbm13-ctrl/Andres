        const canvas = document.getElementById("dungeonCanvas");
        const ctx = canvas.getContext("2d");
        const COLS = 16;

        let dungeonLevel = 1;
        let player = { x: 1, y: 1 };
        let walls = [];
        let gems = [];

        let totalGems = 5;
        let isGameOver = false;

        // Generar nivel procedural validado
        function initDungeonLevel(lvl) {
            dungeonLevel = lvl;
            isGameOver = false;
            isLevelWon = false;
            collectedGems = 0;
            totalGems = 4 + lvl; // Más gemas en niveles altos
            player = { x: 1, y: 1 };
            walls = [];
            gems = [];
            traps = [];

            document.getElementById("dungeonOverlay").classList.add("hidden");
            document.getElementById("dungeonLevel").textContent = dungeonLevel;
            document.getElementById("dungeonScore").textContent = "0";
            document.getElementById("dungeonTotalGems").textContent = totalGems;
            document.getElementById("dungeonStatusMsg").textContent = "¡Recoge todas las gemas doradas!";
            document.getElementById("dungeonStatusMsg").className = "text-emerald-400 font-medium";

            // Muros perimetrales
            for (let r = 0; r < ROWS; r++) {
                for (let c = 0; c < COLS; c++) {
                    if (r === 0 || r === ROWS - 1 || c === 0 || c === COLS - 1) {
                        walls.push({ x: c, y: r });
                    }
                }
            }

            // Obstáculos interiores dispersos (dejando caminos libres)
            const interiorWallCount = Math.min(18, 8 + lvl * 2);
            let attempts = 0;
            while (walls.length < (COLS * 2 + (ROWS - 2) * 2) + interiorWallCount && attempts < 200) {
                attempts++;
                const wx = Math.floor(Math.random() * (COLS - 4)) + 2;
                const wy = Math.floor(Math.random() * (ROWS - 4)) + 2;
                // Evitar bloquear inicio y evitar agrupar demasiado
                if ((wx <= 2 && wy <= 2) || isOccupied(wx, wy)) continue;
                // Evitar encerrar celdas adyacentes
                if (isOccupied(wx + 1, wy) || isOccupied(wx - 1, wy) || isOccupied(wx, wy + 1) || isOccupied(wx, wy - 1)) continue;
                walls.push({ x: wx, y: wy });
            }

            // Colocar Gemas
            attempts = 0;
            while (gems.length < totalGems && attempts < 300) {
                attempts++;
                const gx = Math.floor(Math.random() * (COLS - 2)) + 1;
                const gy = Math.floor(Math.random() * (ROWS - 2)) + 1;
                if ((gx === player.x && gy === player.y) || isOccupied(gx, gy)) continue;
                gems.push({ x: gx, y: gy });
            }
            totalGems = gems.length;
            document.getElementById("dungeonTotalGems").textContent = totalGems;

            // Colocar Trampas de Fuego/Pinchos
            const trapCount = Math.min(12, 3 + lvl * 2);
            attempts = 0;
            while (traps.length < trapCount && attempts < 300) {
                attempts++;
                const tx = Math.floor(Math.random() * (COLS - 2)) + 1;
                const ty = Math.floor(Math.random() * (ROWS - 2)) + 1;
                // Mantener zona segura alrededor del jugador al inicio
                if (Math.abs(tx - player.x) <= 1 && Math.abs(ty - player.y) <= 1) continue;
                if (isOccupied(tx, ty)) continue;
                traps.push({ x: tx, y: ty });
            }

            drawDungeon();
        }

        function isOccupied(x, y) {
            return walls.some(w => w.x === x && w.y === y) ||
                   gems.some(g => g.x === x && g.y === y) ||
                   traps.some(t => t.x === x && t.y === y);
        }

        function movePlayer(dx, dy) {
            if (isGameOver || isLevelWon) return;

            const nx = player.x + dx;
            const ny = player.y + dy;

            // Comprobar colisión con muros
            if (walls.some(w => w.x === nx && w.y === ny)) {
                return;
            }

            player.x = nx;
            player.y = ny;

            // Comprobar colisión con trampas
            if (traps.some(t => t.x === player.x && t.y === player.y)) {
                isGameOver = true;
                showDungeonOverlay(
                    "💀",
                    "¡Caíste en una Trampa!",
                    "Las runas de fuego consumieron tu paso. Intenta cruzar la cámara nuevamente.",
                    "Reintentar Nivel"
                );
                drawDungeon();
                return;
            }

            // Comprobar recolección de gemas
            const gemIndex = gems.findIndex(g => g.x === player.x && g.y === player.y);
            if (gemIndex !== -1) {
                gems.splice(gemIndex, 1);
                collectedGems++;
                document.getElementById("dungeonScore").textContent = collectedGems;

                if (collectedGems >= totalGems) {
                    isLevelWon = true;
                    showDungeonOverlay(
                        "🏆",
                        `¡Cámara ${dungeonLevel} Despejada!`,
                        "Has reunido todos los cristales ancestrales. ¿Listo para descender más profundo?",
                        "Siguiente Cámara"
                    );
                }
            }

            drawDungeon();
        }

        function nextOrRestartDungeon() {
            if (isLevelWon) {
                initDungeonLevel(dungeonLevel + 1);
            } else {
                initDungeonLevel(dungeonLevel);
            }
        }

        // Dibujar escena en Canvas 2D
        function drawDungeon() {
            if (!ctx || canvas.width === 0) return;

            const cellW = canvas.width / COLS;
            const cellH = canvas.height / ROWS;

            // Fondo de losa oscura
            ctx.fillStyle = "#090d18";
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Cuadrícula sutil de baldosas
            ctx.strokeStyle = "rgba(148, 163, 184, 0.06)";
            ctx.lineWidth = 1;
            for (let c = 0; c <= COLS; c++) {
                ctx.beginPath();
                ctx.moveTo(c * cellW, 0);
                ctx.lineTo(c * cellW, canvas.height);
                ctx.stroke();
            }
            for (let r = 0; r <= ROWS; r++) {
                ctx.beginPath();
                ctx.moveTo(0, r * cellH);
                ctx.lineTo(canvas.width, r * cellH);
                ctx.stroke();
            }

            // Dibujar Muros de Piedra Mística
            walls.forEach(w => {
                ctx.fillStyle = "#1e293b";
                ctx.fillRect(w.x * cellW + 1, w.y * cellH + 1, cellW - 2, cellH - 2);
                ctx.strokeStyle = "#334155";
                ctx.strokeRect(w.x * cellW + 1.5, w.y * cellH + 1.5, cellW - 3, cellH - 3);
            });

            // Dibujar Trampas Carmesí
            traps.forEach(t => {
                const cx = (t.x + 0.5) * cellW;
                const cy = (t.y + 0.5) * cellH;
                const radius = Math.min(cellW, cellH) * 0.3;

                ctx.save();
                ctx.fillStyle = "rgba(244, 63, 94, 0.25)";
                ctx.beginPath();
                ctx.arc(cx, cy, radius * 1.25, 0, Math.PI * 2);
                ctx.fill();

                ctx.fillStyle = "#f43f5e";
                ctx.beginPath();
                ctx.moveTo(cx, cy - radius);
                ctx.lineTo(cx + radius, cy + radius * 0.8);
                ctx.lineTo(cx - radius, cy + radius * 0.8);
                ctx.closePath();
                ctx.fill();
                ctx.restore();
            });

            // Dibujar Gemas Doradas con brillo pulsante
            const glowScale = 1 + Math.sin(pulsePhase) * 0.12;
            gems.forEach(g => {
                const cx = (g.x + 0.5) * cellW;
                const cy = (g.y + 0.5) * cellH;
                const r = Math.min(cellW, cellH) * 0.28 * glowScale;

                ctx.save();
                ctx.shadowColor = "#fbbf24";
                ctx.shadowBlur = 12;
                ctx.fillStyle = "#fbbf24";
                ctx.beginPath();
                ctx.moveTo(cx, cy - r);
                ctx.lineTo(cx + r * 0.8, cy);
                ctx.lineTo(cx, cy + r);
                ctx.lineTo(cx - r * 0.8, cy);
                ctx.closePath();
                ctx.fill();
                ctx.restore();
            });

            // Dibujar al Aventurero (Jugador) con aura esmeralda
            const px = (player.x + 0.5) * cellW;
            const py = (player.y + 0.5) * cellH;
            const pr = Math.min(cellW, cellH) * 0.33;

            ctx.save();
            // Aura de antorcha
            const aura = ctx.createRadialGradient(px, py, pr * 0.2, px, py, pr * 2.8);
            aura.addColorStop(0, "rgba(52, 211, 153, 0.35)");
            aura.addColorStop(1, "rgba(52, 211, 153, 0)");
            ctx.fillStyle = aura;
            ctx.beginPath();
            ctx.arc(px, py, pr * 2.8, 0, Math.PI * 2);
            ctx.fill();

            // Cuerpo del explorador
            ctx.fillStyle = "#10b981";
            ctx.beginPath();
            ctx.arc(px, py, pr, 0, Math.PI * 2);
            ctx.fill();
            ctx.lineWidth = 2;
            ctx.strokeStyle = "#ecfdf5";
            ctx.stroke();
            ctx.restore();
        }

        // Bucle de animación suave para el brillo de gemas
        function dungeonLoop() {
            pulsePhase += 0.08;
            drawDungeon();
            animFrameId = requestAnimationFrame(dungeonLoop);
        }

        // Controles de Teclado (WASD y Flechas)
        window.addEventListener("keydown", (e) => {
            // Evitar mover si el usuario está escribiendo en el buscador
            if (document.activeElement === searchInput) return;

            const key = e.key.toLowerCase();
            if (["arrowup", "arrowdown", "arrowleft", "arrowright", "w", "a", "s", "d"].includes(key)) {
                e.preventDefault();
                if (key === "arrowup" || key === "w") movePlayer(0, -1);
                if (key === "arrowdown" || key === "s") movePlayer(0, 1);
                if (key === "arrowleft" || key === "a") movePlayer(-1, 0);
                if (key === "arrowright" || key === "d") movePlayer(1, 0);
            }
        });

        // Gestos táctiles (Swipe) directamente sobre el Canvas
        let touchStartX = 0;
        let touchStartY = 0;

        canvas.addEventListener("touchstart", (e) => {
            touchStartX = e.changedTouches[0].screenX;
            touchStartY = e.changedTouches[0].screenY;
        }, { passive: true });

        canvas.addEventListener("touchend", (e) => {
            const dx = e.changedTouches[0].screenX - touchStartX;
            const dy = e.changedTouches[0].screenY - touchStartY;
            if (Math.abs(dx) < 20 && Math.abs(dy) < 20) return;

            if (Math.abs(dx) > Math.abs(dy)) {
                movePlayer(dx > 0 ? 1 : -1, 0);
            } else {
                movePlayer(0, dy > 0 ? 1 : -1);
            }
        }, { passive: true });

        // Inicialización al cargar la ventana
        window.addEventListener("resize", resizeCanvas);
        window.onload = function() {
            renderCatalog();
            resetRPG();
            initDungeonLevel(1);
            resizeCanvas();
            dungeonLoop();
        };
    </script>
